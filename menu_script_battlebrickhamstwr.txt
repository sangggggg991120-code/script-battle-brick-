local processedChronos = {} 

local function solveChronos()
    local PlayerGui = LocalPlayer:FindFirstChild("PlayerGui")
    if not PlayerGui then return end
    
    for _, gui in ipairs(PlayerGui:GetChildren()) do
        if gui:IsA("ScreenGui") then
            local frame = gui:FindFirstChild("Frame")
            local answerEvent = gui:FindFirstChild("Answer")
            
            if frame and frame:FindFirstChild("Choices") and answerEvent and answerEvent:IsA("RemoteEvent") then
                local choices = frame.Choices
                
                if #choices:GetChildren() > 0 and not processedChronos[gui] then
                    processedChronos[gui] = true 
                    
                    task.wait(ClockWaitTime)
                    
                    if gui and gui.Parent and choices then
                        pcall(function()
                            answerEvent:FireServer(true)[span_0](start_span)[span_0](end_span)
                            
                            for _, asset in ipairs(gui:GetDescendants()) do
                                if asset:IsA("ImageButton") or asset:IsA("ImageLabel") then
                                    TweenService:Create(asset, TweenInfo.new(1.5), {
                                        ["ImageTransparency"] = 1,
                                        ["BackgroundTransparency"] = 1
                                    }):Play()
                                    asset.Active = false
                                    asset.Selectable = false
                                elseif asset:IsA("Frame") then
                                    TweenService:Create(asset, TweenInfo.new(1.5), {
                                        ["BackgroundTransparency"] = 1
                                    }):Play()
                                end
                            end
                            game:GetService("Debris"):AddItem(gui, 2)[span_1](start_span)[span_1](end_span)
                        end)
                    end
                    
                    gui.Destroying:Connect(function()
                        processedChronos[gui] = nil
                    end)
                    break
                end
            end
        end
    end
end

task.spawn(function()
    while true do
        if ChronoEnabled then
            pcall(solveChronos)
        end
        task.wait(0.5) 
    end
end)

-- CORE LOGIC 2: AIMBOT AUTO SHOT FUNK1D[NORMAL]
-- ==========================================
local TargetQueue = {}
local Processing = false
local LoggedTargets = {}
local CurrentShootEvent = nil

local function isLegitTarget(obj)
    if not obj or not obj:IsDescendantOf(workspace) then return false end
    if LoggedTargets[obj] then return false end
    
    local targetType = obj:GetAttribute("Type")
    if targetType then
        return tostring(targetType) == "Target"
    end
    
    local name = string.lower(obj.Name)
    if string.find(name, "bomb") or string.find(name, "bom") or 
       string.find(name, "cross") or string.find(name, "fake") or 
       string.find(name, "hazard") then
        return false
    end
    return true
end

local function getActiveShootEvent()
    if CurrentShootEvent and CurrentShootEvent:IsDescendantOf(game) then
        return CurrentShootEvent
    end
    
    local playerGui = LocalPlayer:FindFirstChild("PlayerGui")
    if playerGui then
        for _, v in ipairs(playerGui:GetDescendants()) do
            if v.Name == "ShootEvent" and v:IsA("RemoteEvent") then
                CurrentShootEvent = v
                table.clear(LoggedTargets)
                table.clear(TargetQueue)
                return CurrentShootEvent
            end
        end
    end
    return nil
end

local function processQueue()
    if Processing then return end
    Processing = true
    
    while #TargetQueue > 0 do
        if not FunkEnabled then
            table.clear(TargetQueue)
            break
        end

        local shootEvent = getActiveShootEvent()
        if not shootEvent then 
            task.wait(1)
            break 
        end
        
        local target = table.remove(TargetQueue, 1)
        if target and target:IsDescendantOf(workspace) then
            shootEvent:FireServer(target)
            task.wait(FunkWaitTime)
        end
    end
    Processing = false
end

local function enqueueTarget(target)
    if not FunkEnabled then return end
    if isLegitTarget(target) then
        LoggedTargets[target] = true
        table.insert(TargetQueue, target)
        task.spawn(processQueue)
    end
end

CollectionService:GetInstanceAddedSignal("Target"):Connect(function(target)
    enqueueTarget(target)
end)

task.spawn(function()
    while true do
        if FunkEnabled and getActiveShootEvent() then
            for _, target in ipairs(CollectionService:GetTagged("Target")) do
                enqueueTarget(target)
            end
        end
        task.wait(1)
    end
end)

-- LOGIC DI CHUYỂN CỤC VUÔNG NHỎ QUA RUNSERVICE (HẾT KHỰNG)
local dragging = false
local dragInput = nil
local dragStart = nil
local startPos = nil

local function update(input)
    local delta = input.Position - dragStart
    local newPos = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
    MiniBtn.Position = newPos
    savedMiniPosition = newPos
end

MiniBtn.InputBegan:Connect(function(input)
    if (input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch) then
        dragging = true
        dragInput = input
        dragStart = input.Position
        startPos = MiniBtn.Position
        
        input.Changed:Connect(function()
            if input.UserInputState == Enum.UserInputState.End then
                dragging = false
            end
        end)
    end
end)

UserInputService.InputChanged:Connect(function(input)
    if input == dragInput and dragging then
        update(input)
    end
end)
